    private void BtnGuardarPedidojButtonGuardarProducto(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_BtnGuardarPedidojButtonGuardarProducto
        // TODO add your handling code here:
        //Validación
        if(tblProductosPedido.getRowCount() == 0){

            JOptionPane.showMessageDialog(this,
                    "Agrega al menos un producto");

            return;

        }
        
        try{

            Pedido ped = new Pedido();

            ped.setCliente(
                    txtCliente.getText());

            SimpleDateFormat formato =
                    new SimpleDateFormat("yyyy-MM-dd");

            ped.setFechaInicio(
                    formato.parse(
                            txtFechaInicio.getText()));

            ped.setFechaEntrega(
                    formato.parse(
                            txtFechaEntrega.getText()));

            ped.setObservaciones(
                    txtObservaciones.getText());

            int idEstado =
                    cmbEstadoPedido.getSelectedIndex();

            ped.setIdEstadoPedido(idEstado);

            ProduccionDAO dao = new ProduccionDAO();
            //Guardamos en detalle de pedido
            int idPedido =
                dao.guardarPedido(ped);

            if(idPedido > 0){

                DefaultTableModel modelo =
                        (DefaultTableModel) tblProductosPedido.getModel();

                for(int i = 0; i < modelo.getRowCount(); i++){

                    int idProducto =
                            Integer.parseInt(
                                    modelo.getValueAt(i, 0).toString());

                    int cantidad =
                            Integer.parseInt(
                                    modelo.getValueAt(i, 2).toString());

                    dao.guardarDetallePedido(
                            idPedido,
                            idProducto,
                            cantidad);

                }

                JOptionPane.showMessageDialog(this,
                        "Pedido guardado correctamente");
                //Actualizar la tabla
                dao.mostrarPedidos(tblPedidos);

            }else{

                JOptionPane.showMessageDialog(this,
                        "Error al guardar");

            }
        }catch(Exception e){
            System.out.println("Error: " + e);
        }
        
        limpiar();
    }//GEN-LAST:event_BtnGuardarPedidojButtonGuardarProducto

    private void buttonEditarActionPerformedActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_buttonEditarActionPerformedActionPerformed
        // TODO add your handling code here:}
        
        if(txtIdProduccion.getText().equals("")){

            JOptionPane.showMessageDialog(this,
                    "Selecciona un pedido");

            return;

        }    
        
        int confirmar = JOptionPane.showConfirmDialog(this, "¿Estás seguro de que deseas actualizar este pedido?",
            "Confirmar actualización", JOptionPane.YES_NO_OPTION);

        if(confirmar != JOptionPane.YES_OPTION){

            return;
        }
        
        try{
            Pedido ped = new Pedido();

            ped.setIdPedido(Integer.parseInt(txtIdProduccion.getText()));

            ped.setCliente(txtCliente.getText());

            SimpleDateFormat formato =
            new SimpleDateFormat("yyyy-MM-dd");

            ped.setFechaEntrega(formato.parse(txtFechaEntrega.getText()));

            ped.setObservaciones(txtObservaciones.getText());

            int idEstado =cmbEstadoPedido.getSelectedIndex();

            ped.setIdEstadoPedido(idEstado);

            ProduccionDAO dao = new ProduccionDAO();

            boolean respuesta = dao.editarPedido(ped);

            if(respuesta){

                dao.eliminarDetallePedido(ped.getIdPedido());

                DefaultTableModel modelo = (DefaultTableModel) tblProductosPedido.getModel();

                for(int i = 0; i < modelo.getRowCount(); i++){

                    int idProducto = Integer.parseInt(modelo.getValueAt(i, 0).toString());

                    int cantidad = Integer.parseInt(modelo.getValueAt(i, 2).toString());

                    dao.guardarDetallePedido(ped.getIdPedido(),idProducto,cantidad);
                }

                JOptionPane.showMessageDialog(this,
                        "Pedido actualizado");

                dao.mostrarPedidos(tblPedidos);

            }else{

                JOptionPane.showMessageDialog(this,
                        "Error al actualizar");

            }
        }catch(Exception e){

        System.out.println("Error: " + e);

        }
    }//GEN-LAST:event_buttonEditarActionPerformedActionPerformed

    private void ButtonEliminarActionPerformedActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_ButtonEliminarActionPerformedActionPerformed
        // TODO add your handling code here:
        if(txtIdProduccion.getText().equals("")){

            JOptionPane.showMessageDialog(this,
                    "Selecciona un pedido");

            return;

        }    
        
        int confirmar = JOptionPane.showConfirmDialog(this, "¿Estás seguro de que deseas eliminar este pedido?",
            "Confirmar eliminación", JOptionPane.YES_NO_OPTION);

        if(confirmar != JOptionPane.YES_OPTION){

            return;
        }

        int idPedido =
                Integer.parseInt(
                        txtIdProduccion.getText());

        ProduccionDAO dao =
                new ProduccionDAO();

        dao.eliminarDetallePedido(idPedido);

        boolean respuesta =
                dao.eliminarPedido(idPedido);

        if(respuesta){

            JOptionPane.showMessageDialog(this,
                    "Pedido eliminado");

            dao.mostrarPedidos(tblPedidos);

            limpiar();

        }else{

            JOptionPane.showMessageDialog(this,
                    "Error al eliminar");

        }
        //Actualizar la tabla
        dao.mostrarPedidos(tblPedidos);
        
        limpiar();
    }//GEN-LAST:event_ButtonEliminarActionPerformedActionPerformed

    private void BtnAgregarProductoPedidoActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_BtnAgregarProductoPedidoActionPerformed
         // 1. Obtenemos el texto y quitamos espacios en blanco extra con trim()
    String textoCantidad = txtCantidad.getText().trim();
    
    // 2. Validamos que no esté vacío
    if (textoCantidad.isEmpty()) {
        javax.swing.JOptionPane.showMessageDialog(this, "Por favor, ingresa una cantidad.", "Campo vacío", javax.swing.JOptionPane.WARNING_MESSAGE);
        return; // Detiene la ejecución para que no avance al error
    }

    int cantidad = 0;
    
    // 3. Validamos que realmente sea un número entero
    try {
        cantidad = Integer.parseInt(textoCantidad);
        
        // Validamos que no ingresen números negativos o cero si tu lógica lo requiere
        if (cantidad <= 0) {
            javax.swing.JOptionPane.showMessageDialog(this, "La cantidad debe ser mayor a cero.", "Valor inválido", javax.swing.JOptionPane.WARNING_MESSAGE);
            return;
        }
        
    } catch (NumberFormatException e) {
        javax.swing.JOptionPane.showMessageDialog(this, "La cantidad debe ser un número entero válido (Ej: 1, 2, 5).", "Error de formato", javax.swing.JOptionPane.ERROR_MESSAGE);
        return; // Detiene la ejecución
    }

    // 4. Si todo salió bien arriba, continuamos con tu lógica original
    String producto = cmbNombreProductos.getSelectedItem().toString();
    ProductoDAO dao = new ProductoDAO();
    
    int idProducto = dao.obtenerIdProducto(producto);
    double precio = dao.obtenerPrecioProducto(producto);
    double subtotal = cantidad * precio;

    DefaultTableModel modelo = (DefaultTableModel) tblProductosPedido.getModel();
    modelo.addRow(new Object[]{
        idProducto,
        producto,
        cantidad,
        precio,
        subtotal
    });
    
    // Suma los subtotales de los productos seleccionados
    calcularTotalPedido();
    
    // Limpiar campos
    txtCantidad.setText("");
    cmbNombreProductos.setSelectedIndex(0);
    }//GEN-LAST:event_BtnAgregarProductoPedidoActionPerformed

    private void BtnQuitarProductoPedidoActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_BtnQuitarProductoPedidoActionPerformed

    }//GEN-LAST:event_BtnQuitarProductoPedidoActionPerformed

   public void calcularTotalPedido(){

    double total = 0;

    DefaultTableModel modelo =
            (DefaultTableModel) tblProductosPedido.getModel();

    for(int i = 0; i < modelo.getRowCount(); i++){

        total += Double.parseDouble(
                modelo.getValueAt(i, 4).toString());

    }

    txtTotalPedido.setText(
            String.valueOf(total));

    }
