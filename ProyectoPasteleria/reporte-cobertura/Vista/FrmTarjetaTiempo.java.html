    private void AgregarTiempoActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_AgregarTiempoActionPerformed
        // TODO add your handling code here:
        try{

            LocalTime inicio = LocalTime.parse(txtHoraInicio.getText());

            LocalTime fin = LocalTime.parse(txtHoraFin.getText());

            Duration duracion = Duration.between(inicio, fin);

            double horas = duracion.toMinutes() / 60.0;

            txtTiempoTotal.setText( String.valueOf(horas));

            double tasaHora = Double.parseDouble(txtTasaHora.getText());

            double costo = horas * tasaHora;

            txtCostoTotal.setText(String.format("%.2f", costo));

            DefaultTableModel modelo =(DefaultTableModel)tblTiempos.getModel();

            modelo.addRow(new Object[]{

                modelo.getRowCount() + 1,
                txtFechaTarjeta.getText(),
                txtHoraInicio.getText(),
                txtHoraFin.getText(),
                horas,
                tasaHora,
                String.format("%.2f", costo)

            });
            
            double totalHoras = 0;
            double totalCosto = 0;

            for(int i = 0;
                i < modelo.getRowCount();
                i++){

                totalHoras += Double.parseDouble(
                        modelo.getValueAt(i, 4)
                        .toString());

                totalCosto += Double.parseDouble(
                        modelo.getValueAt(i, 6)
                        .toString());

            }

            txtTotalHoras.setText(
                    String.format("%.2f",
                            totalHoras));

            txtTotalCosto.setText(
                    String.format("%.2f",
                            totalCosto));

        }catch(Exception e){

            JOptionPane.showMessageDialog(this,
                    "Verifica las horas.\n" + "Formato correcto: HH:mm");

            System.out.println("Error: " + e);
        }
    
        
    }//GEN-LAST:event_AgregarTiempoActionPerformed

    private void BtnGuardarTarjetajButtonGuardarEmpleado(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_BtnGuardarTarjetajButtonGuardarEmpleado
        // TODO add your handling code here:
        try{

            TarjetaTiempo tarjeta = new TarjetaTiempo();

            TarjetaTiempoDAO dao = new TarjetaTiempoDAO();

            // ID EMPLEADO

            tarjeta.setIdEmpleado(
                    Integer.parseInt(
                            txtIdEmpleado.getText()));

            // ID PEDIDO

            String pedido =
                    cmbNoPedido
                    .getSelectedItem()
                    .toString();

            int idPedido =
                    Integer.parseInt(
                            pedido.split(" - ")[0]);

            tarjeta.setIdPedido(
                    idPedido);

            // FECHA

            SimpleDateFormat formato = new SimpleDateFormat("yyyy-MM-dd");

            tarjeta.setFecha(formato.parse(txtFechaTarjeta.getText()));

            // OBSERVACIONES

            tarjeta.setObservaciones(txtDescripcion.getText());

            // TOTALES

            tarjeta.setTotalHoras(Double.parseDouble(txtTotalHoras.getText()));

            tarjeta.setTotalCosto(Double.parseDouble(txtTotalCosto.getText()));

            // GUARDAR ENCABEZADO

            int idTarjeta =dao.guardarTarjetaTiempo(tarjeta);

            if(idTarjeta > 0){

                DefaultTableModel modelo = (DefaultTableModel)tblTiempos.getModel();

                for(int i = 0; i < modelo.getRowCount(); i++){

                    String dia =modelo.getValueAt(i,1).toString();

                    String horaInicio = modelo.getValueAt(i,2).toString();

                    String horaFin = modelo.getValueAt(i,3).toString();

                    double tiempoTotal = Double.parseDouble(modelo.getValueAt(i,4).toString());

                    double tasaHora = Double.parseDouble(modelo.getValueAt(i,5).toString());

                    double costoTotal = Double.parseDouble(modelo.getValueAt(i,6).toString());

                    dao.guardarDetalleTiempo(idTarjeta,dia,horaInicio,horaFin,tiempoTotal,tasaHora,costoTotal);

                }

                JOptionPane.showMessageDialog(this,"Tarjeta de tiempo guardada correctamente");
                dao.mostrarTarjetas(tblTarjetaTiempo);

            }else{

                JOptionPane.showMessageDialog(this,"Error al guardar");

            }

        }catch(Exception e){

            System.out.println("Error: " + e);

        }

    }//GEN-LAST:event_BtnGuardarTarjetajButtonGuardarEmpleado

    private void buttonEditarActionPerformedActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_buttonEditarActionPerformedActionPerformed
        // TODO add your handling code here:

            return;

    private void ButtonEliminarActionPerformedActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_ButtonEliminarActionPerformedActionPerformed
        // TODO add your handling code here:
        //CONFIRMAR ELIMINACIÓN DE LA FILA
        int respuesta =JOptionPane.showConfirmDialog(this,
                "¿Deseas eliminar el registro?",
                "Confirmar",
                JOptionPane.YES_NO_OPTION);
        
        int fila = tblTiempos.getSelectedRow();

        if(fila == -1){

            JOptionPane.showMessageDialog(this,"Selecciona un registro");

            return;

        }

        DefaultTableModel modelo = (DefaultTableModel)tblTiempos.getModel();

        modelo.removeRow(fila);
        
        //CONTINUAR LA NUMERACIÓN
        for(int i = 0; i < modelo.getRowCount(); i++){

            modelo.setValueAt(i + 1, i, 0);

        }

        // RECALCULAR TOTALES

        double totalHoras = 0;
        double totalCosto = 0;

        for(int i = 0; i < modelo.getRowCount(); i++){

            totalHoras += Double.parseDouble(modelo.getValueAt(i,4).toString());

            totalCosto += Double.parseDouble(modelo.getValueAt(i,6).toString());

        }

        txtTotalHoras.setText(String.format("%.2f",totalHoras));

        txtTotalCosto.setText(String.format("%.2f",totalCosto));

    
    }//GEN-LAST:event_ButtonEliminarActionPerformedActionPerformed

    private void tblTarjetaTiempoMouseClicked(java.awt.event.MouseEvent evt) {//GEN-FIRST:event_tblTarjetaTiempoMouseClicked
        // TODO add your handling code here:
        
        //OBTENER EL ID DE LA TARJETA QUE SE SELECCIONA
        int fila = tblTarjetaTiempo.getSelectedRow();

        if(fila == -1){

            return;

        }

        int idTarjeta = Integer.parseInt(tblTarjetaTiempo.getValueAt(fila, 0).toString());

        TarjetaTiempoDAO dao = new TarjetaTiempoDAO();

        dao.mostrarDetalleTarjeta(tblTiempos, idTarjeta);
        TarjetaTiempo tarjeta = dao.obtenerTarjetaTiempo(idTarjeta);
        
        txtNoTarjeta.setText(String.valueOf(tarjeta.getIdTarjeta()));

        txtIdEmpleado.setText(String.valueOf(tarjeta.getIdEmpleado()));

        txtFechaTarjeta.setText(tarjeta.getFecha().toString());

        txtDescripcion.setText(tarjeta.getObservaciones());

        txtTotalHoras.setText(String.valueOf(tarjeta.getTotalHoras()));

        txtTotalCosto.setText(String.valueOf(tarjeta.getTotalCosto()));
        
        for(int i = 0;
            i < cmbEmpleado.getItemCount();
            i++){

            String item =
                    cmbEmpleado
                    .getItemAt(i)
                    .toString();

            int idEmpleadoCombo =
                    Integer.parseInt(
                            item.split(" - ")[0]);

            if(idEmpleadoCombo ==
                    tarjeta.getIdEmpleado()){

                cmbEmpleado.setSelectedIndex(i);

                break;

            }

        }
        
        for(int i = 0;
            i < cmbNoPedido.getItemCount();
            i++){

            String item =
                    cmbNoPedido
                    .getItemAt(i)
                    .toString();

            int idPedidoCombo =
                    Integer.parseInt(
                            item.split(" - ")[0]);

            if(idPedidoCombo ==
                    tarjeta.getIdPedido()){

    private void BtnImprimirActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_BtnImprimirActionPerformed
        // TODO add your handling code here:
        try{

            JTextArea area =  new JTextArea();

            area.append(
                    "******** TARJETA DE TIEMPO ********\n\n");

            area.append(
                    "No. Tarjeta: "
                    + txtNoTarjeta.getText()
                    + "\n");

            area.append(
                    "Empleado: "
                    + cmbEmpleado.getSelectedItem()
                    + "\n");

            area.append(
                    "No. Nomina: "
                    + txtIdEmpleado.getText()
                    + "\n");

            area.append(
                    "Pedido: "
                    + cmbNoPedido.getSelectedItem()
                    + "\n");

            area.append(
                    "Fecha: "
                    + txtFechaTarjeta.getText()
                    + "\n\n");

            area.append(
                    "Observaciones: "
                    + txtDescripcion.getText()
                    + "\n\n");

            area.append(
                    "------------------------------------------------------------\n");

            area.append(
                    String.format(
                            "%-12s %-10s %-10s %-10s %-10s %-10s\n",
                            "Dia",
                            "Inicio",
                            "Fin",
                            "Horas",
                            "Tasa",
                            "Costo"));

            area.append(
                    "------------------------------------------------------------\n");

            DefaultTableModel modelo =
                    (DefaultTableModel)
                    tblTiempos.getModel();

            for(int i = 0;
                i < modelo.getRowCount();
                i++){

                area.append(
                        String.format(
                                "%-12s %-10s %-10s %-10s %-10s %-10s\n",

                                modelo.getValueAt(i,1),
                                modelo.getValueAt(i,2),
                                modelo.getValueAt(i,3),
                                modelo.getValueAt(i,4),
                                modelo.getValueAt(i,5),
                                modelo.getValueAt(i,6)));

            }

            area.append(
                    "\n------------------------------------------------------------\n");

            area.append(
                    "Total Horas: "
                    + txtTotalHoras.getText()
                    + "\n");

            area.append(
                    "Total Costo: $"
                    + txtTotalCosto.getText()
                    + "\n");

            area.print();

        }catch(Exception e){

            JOptionPane.showMessageDialog(
                    this,
                    "Error al imprimir: "
                    + e.getMessage());

        }
    }//GEN-LAST:event_BtnImprimirActionPerformed
